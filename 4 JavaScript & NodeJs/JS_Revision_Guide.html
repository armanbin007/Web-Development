1. Variables, Data Types & Objects

Topic 55
  • var → function-scoped, hoisted, avoid using in modern JS.
  • let → block-scoped, can be reassigned.
  • const → block-scoped, cannot be reassigned (but object contents can still change).
  • A block { } creates a new scope for let/const — that's why the inner a doesn't clash with the outer one.
script.js
let a = 35;
let b = 20;

{
  let a = 900; // shadows outer 'a' only inside this block
  console.log("inner a:", a);
}
console.log("outer a:", a);

let o = {
  "name": "Programmer",
  "job code": 530,
  "area": "Europe"
};
console.log(o);
o.salary = "1 lac"; // objects can gain new props even with const/let
console.log(o);
Key idea: object properties can always be added/changed even when the object itself was declared with const — const only locks the variable binding, not the contents.

7. Recursion — Factorial

Topic 65

A recursive function calls itself with a smaller input until it hits a base case that stops it.

script.js
function factorial(x) {
  if (x < 0) return;
  if (x <= 1) return 1;      // base case
  return x * factorial(x - 1); // recursive case
}
console.log("5! =", factorial(5));

// Iterative alternative using reduce
function fac(x) {
  let arr = Array.from(Array(x + 1).keys()); // [0,1,2,...,x]
  return arr.slice(1).reduce((a, b) => a * b);
}
console.log("6! (iterative) =", fac(6));

9. Events & Bubbling

10. Callbacks & Promises

Topic 75
  • A callback is a function passed into another function to be run later (e.g. after a script loads).
  • A Promise represents a value that will exist eventually — it's pending, then settles as resolved or rejected.
  • .then() handles success, .catch() handles failure.
  • Promise.all([...]) — waits for all, rejects if any one rejects.
  • Promise.any([...]) — resolves as soon as any one resolves.
  • Promise.race([...]) — settles as soon as the first one settles (win or lose).
  • Promise.allSettled([...]) — waits for all, never short-circuits, gives status of each.
promise.js
let prom1 = new Promise((resolve, reject) => {
  setTimeout(() => resolve("Done! (prom1)"), 300);
});
let prom2 = new Promise((resolve, reject) => {
  setTimeout(() => resolve("Done! (prom2)"), 100);
});

Promise.any([prom1, prom2])
  .then((a) => console.log("First to resolve:", a))
  .catch((e) => console.log("error:", e));

11. Async/Await & Fetch API

Topic 76
  • async function always returns a Promise.
  • await pauses execution inside an async function until the Promise settles.
  • fetch(url, options) makes an HTTP request; await res.json() or .text() reads the body.
script.js
function sleep(ms, value) {
  return new Promise(resolve => setTimeout(() => resolve(value), ms));
}

async function runX() {
  console.log("Fetching data...");
  let data = await sleep(300, "Data received"); // waits before continuing
  console.log(data);
  console.log("Process finished");
}
runX();
Your actual API version used fetch('https://jsonplaceholder.typicode.com/posts', { method: 'POST', body: JSON.stringify({...}), headers: {'Content-type':'application/json'} }) — same idea, just a real network call instead of sleep().

12. Try / Catch / Finally

Topic 79
  • try — code that might throw an error.
  • catch (error) — runs only if something inside try throws.
  • finally — always runs, error or not (good for cleanup).
  • throw new Error("msg") — manually raise an error.
script.js
function lama(sum) {
  try {
    if (isNaN(sum)) {
      throw new Error("Please enter valid numbers");
    }
    console.log(`The sum is: ${sum}`);
    return true;
  } catch (error) {
    console.log("Error:", error.message);
    return false;
  } finally {
    console.log("Files are being closed");
  }
}
lama(15);
lama(NaN);

14. Destructuring, Spread/Rest & IIFE

Topic 82
  • Array destructuring: let [x, y, ...rest] = arr
  • Object destructuring: let {a, b} = obj — pulls named properties out.
  • Spread ...arr — expands an array into individual elements (e.g. as function args).
  • Rest — same ... syntax but collects remaining items into an array.
  • IIFE (Immediately Invoked Function Expression): (function(){...})() — defines and runs a function instantly, often to create a private scope or to use await at the top level.
Destructuring.js + IIFE.js
let [x, y, ...rest] = [1, 5, 7, 8, 9];
console.log(x, y, rest);

let obj = { a: 60, b: 98, c: 10 };
let { a, b } = obj;
console.log(a, b);

function sum(a, b, c) { return a + b + c; }
let arr = [1, 5, 6];
console.log(sum(...arr)); // spread expands array into 3 args

function sleep() {
  return new Promise(resolve => setTimeout(() => resolve(45), 200));
}

(async function main() { // IIFE so we can use await immediately
  let result = await sleep();
  console.log("IIFE result:", result);
})();
